function loadContent(page) {
  const contentArea = document.querySelector('.content');
  
  // Clear previous content
  contentArea.innerHTML = '';
  
  // Create HTML for selected page
  switch(page) {
    case 'members':
      contentArea.innerHTML = getMembersContent();
      break;
    case 'bookings':
      contentArea.innerHTML = getBookingsContent();
      break;
    case 'video':
      contentArea.innerHTML = getVideoContent();
      break;
    case 'messages':
      contentArea.innerHTML = getMessagesContent();
      break;
    case 'registrations':
      contentArea.innerHTML = getRegistrationsContent();
      break;
    case 'admin':
      contentArea.innerHTML = getAdminContent();
      break;
    case 'rules':
      contentArea.innerHTML = getRulesContent();
      break;
    case 'news':
      contentArea.innerHTML = getNewsContent();
      break;
    default:
      contentArea.innerHTML = '<div class="error">Страница не найдена</div>';
  }
}

// Menu item click handler
document.querySelectorAll('.sidebar a').forEach(link => {
  link.addEventListener('click', (e) => {
    e.preventDefault();
    const page = link.getAttribute('data-page');
    loadContent(page);
    closeSidebar();
  });
});
